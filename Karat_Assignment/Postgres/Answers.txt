

(a) Products with category and supplier names
SELECT p.product_name, c.category_name, s.supplier_name
FROM products p
JOIN categories c ON p.category_id = c.category_id
JOIN suppliers s ON p.supplier_id = s.supplier_id;

(b) Products with stock below 5
SELECT *
FROM products
WHERE stock_quantity < 5;

(c) Add discount_percent column
ALTER TABLE products
ADD COLUMN discount_percent NUMERIC(5,2) DEFAULT 0;

(d) Reduce Electronics prices by 15%
UPDATE products
SET price = price * 0.85
WHERE category_id = (
    SELECT category_id
    FROM categories
    WHERE category_name = 'Electronics'
);

(e) Total number of products
SELECT COUNT(*) AS total_products
FROM products;

(f) Average price
SELECT AVG(price) AS avg_price
FROM products;

(g) Max and Min Electronics price
SELECT MAX(p.price) AS max_price,
       MIN(p.price) AS min_price
FROM products p
JOIN categories c ON p.category_id = c.category_id
WHERE c.category_name = 'Electronics';

(h) Categories with product count
SELECT c.category_name,
       COUNT(p.product_id) AS product_count
FROM categories c
LEFT JOIN products p ON c.category_id = p.category_id
GROUP BY c.category_name;

(i) Suppliers supplying products between $50 and $200
SELECT DISTINCT s.supplier_name
FROM suppliers s
JOIN products p ON s.supplier_id = p.supplier_id
WHERE p.price BETWEEN 50 AND 200;

(j) Products in category IDs 1 and 3
SELECT *
FROM products
WHERE category_id IN (1,3);

(k) Total stock per category having more than 1 product
SELECT c.category_name,
       SUM(p.stock_quantity) AS total_stock
FROM categories c
JOIN products p ON c.category_id = p.category_id
GROUP BY c.category_name
HAVING COUNT(*) > 1;

(l) Suppliers with average product price > 100
SELECT s.supplier_name,
       AVG(p.price) AS avg_price
FROM suppliers s
JOIN products p ON s.supplier_id = p.supplier_id
GROUP BY s.supplier_name
HAVING AVG(p.price) > 100;

(m) Products sorted by price descending
SELECT *
FROM products
ORDER BY price DESC;

(n) Total stock value per category
SELECT c.category_name,
       SUM(p.price * p.stock_quantity) AS total_stock_value
FROM categories c
JOIN products p ON c.category_id = p.category_id
GROUP BY c.category_name
ORDER BY total_stock_value DESC;

JOINS

(a) INNER JOIN
SELECT p.product_name, c.category_name
FROM products p
INNER JOIN categories c ON p.category_id = c.category_id;

(b) LEFT JOIN
SELECT p.product_name, c.category_name
FROM products p
LEFT JOIN categories c ON p.category_id = c.category_id;

(c) Categories with count of products
SELECT c.category_name,
       COUNT(p.product_id) AS product_count
FROM categories c
LEFT JOIN products p ON c.category_id = p.category_id
GROUP BY c.category_name;

(d) Products with supplier names
SELECT p.product_name,
       s.supplier_name
FROM products p
LEFT JOIN suppliers s ON p.supplier_id = s.supplier_id;

(e) Suppliers and their products (RIGHT JOIN)
SELECT s.supplier_name,
       p.product_name
FROM products p
RIGHT JOIN suppliers s ON p.supplier_id = s.supplier_id;

(f) Products without supplier
SELECT *
FROM products
WHERE supplier_id IS NULL;

(g) Products with category and supplier
SELECT p.product_name,
       c.category_name,
       s.supplier_name
FROM products p
JOIN categories c ON p.category_id = c.category_id
JOIN suppliers s ON p.supplier_id = s.supplier_id;

(h) Suppliers and categories
SELECT s.supplier_name,
       c.category_name
FROM suppliers s
CROSS JOIN categories c;

(i) Products whose supplier email is not NULL
SELECT p.product_name,
       s.contact_email
FROM products p
JOIN suppliers s ON p.supplier_id = s.supplier_id
WHERE s.contact_email IS NOT NULL;

(j) Categories having products supplied by Global Goods
SELECT DISTINCT c.category_name
FROM categories c
JOIN products p ON c.category_id = p.category_id
JOIN suppliers s ON p.supplier_id = s.supplier_id
WHERE s.supplier_name = 'Global Goods';

BONUS WINDOW FUNCTIONS

SELECT product_name,
       price,
       RANK() OVER(ORDER BY price DESC) AS rank_no
FROM products;

SELECT product_name,
       price,
       DENSE_RANK() OVER(ORDER BY price DESC) AS dense_rank_no
FROM products;

SELECT *
FROM (
    SELECT p.*,
           ROW_NUMBER() OVER(PARTITION BY category_id ORDER BY price DESC) rn
    FROM products p
) x
WHERE rn = 1;
