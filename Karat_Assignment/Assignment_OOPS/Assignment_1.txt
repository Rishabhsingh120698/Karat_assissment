abstract class Vehicle {

    int speed = 100;
    long distance = 500;

    
    Vehicle() {
        System.out.println("Vehicle default constructor called");
    }

    
    Vehicle(int speed, long distance) {
        this.speed = speed;
        this.distance = distance;
    }

   
    abstract void run();

    abstract void stop();

    
    public void fuel(int quantity) {
        System.out.println("Fuel quantity = " + quantity);
    }

    public void fuel(float quantity, String type) {
        System.out.println("Fuel quantity = " + quantity);
        System.out.println("Fuel type = " + type);
    }

    public void fuel(char type, int quantity) {
        System.out.println("Fuel type = " + type);
        System.out.println("Fuel quantity = " + quantity);
    }
}


// ===============================
// Two Wheeler
// ===============================

class TwoWheeler extends Vehicle {

    int speed = 80;
    long distance = 300;
    int nos_of_tyre = 2;

   
    TwoWheeler() {
        super();
    }

    @Override
    void run() {
        System.out.println("Two Wheeler is running");
    }

    @Override
    void stop() {
        System.out.println("Two Wheeler is stopped");
    }

    public void display() {

        System.out.println("Two Wheeler Details");

        System.out.println("Speed = " + speed);
        System.out.println("Distance = " + distance);
        System.out.println("Number of Tyres = " + nos_of_tyre);

        System.out.println("Parent Speed = " + super.speed);
        System.out.println("Parent Distance = " + super.distance);
    }
}
